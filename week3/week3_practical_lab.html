1. Caesar and brute force

Concept: C = (P + K) mod 26. Brute force works because there are only 26 keys.

A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

def caesar_enc(text, k):
    """C = (P + K) mod 26"""
    return "".join(A[(A.index(c) + k) % 26] for c in text.upper() if c in A)

def caesar_dec(text, k):
    """P = (C - K) mod 26"""
    return caesar_enc(text, -k)

def brute_force(ciphertext):
    """Try every key. Works because the key space is tiny."""
    for k in range(26):
        print(k, caesar_dec(ciphertext, k))

# --- tests (silent if everything is correct) ---

assert caesar_enc("RIVER", 39) == "EVIRE"          # 39 = 13 mod 26
assert caesar_dec(caesar_enc("HELLO", 5), 5) == "HELLO"

2. Vigenere

Concept: a different Caesar shift per position, cycling through the key with i % len(key).

A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

def vig(text, key, direction=1):
    """direction = +1 encrypt, -1 decrypt.  Ci = (Pi + K[i mod M]) mod 26"""
    text = [c for c in text.upper() if c in A]
    key = [c for c in key.upper() if c in A]
    return "".join(
        A[(A.index(p) + direction * A.index(key[i % len(key)])) % 26]
        for i, p in enumerate(text)
    )

def vig_enc(pt, key): return vig(pt, key, +1)
def vig_dec(ct, key): return vig(ct, key, -1)

# --- tests: examples taken from the lecture slides ---

assert vig_enc("WEAREDISCOVEREDSAVEYOURSELF", "DECEPTIVE") == "ZICVTWQNGRZGVTWAVZHCQYGLMGJ"
assert vig_enc("HIDDEN", "KEY") == "RMBNIL"
assert vig_dec("RMBNIL", "KEY") == "HIDDEN"

3. Playfair

Concept: build a 5x5 grid, split into digraphs, apply row, column or rectangle rule.

A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

def pf_matrix(keyword):
    """5x5 grid: keyword letters (no duplicates) then the rest of the alphabet. I and J share a cell."""
    seen = []
    for c in (keyword.upper() + A).replace("J", "I"):
        if c in A and c not in seen:
            seen.append(c)
    return [seen[i * 5:(i + 1) * 5] for i in range(5)]

def pf_pos(m, ch):
    for r in range(5):
        for c in range(5):
            if m[r][c] == ch:
                return r, c

def pf_digraphs(text, filler="X"):
    """Split into pairs. A repeated pair gets a filler; pad the end if odd."""
    t = [c for c in text.upper().replace("J", "I") if c in A]
    pairs, i = [], 0
    while i < len(t):
        a = t[i]
        b = t[i + 1] if i + 1 < len(t) else filler
        if a == b:
            pairs.append(a + filler)
            i += 1
        else:
            pairs.append(a + b)
            i += 2
    return pairs

def pf_crypt(pairs, m, d):
    """d = +1 encrypt (right / down), d = -1 decrypt (left / up)."""
    out = []
    for p in pairs:
        (r1, c1), (r2, c2) = pf_pos(m, p[0]), pf_pos(m, p[1])
        if r1 == r2:                                   # same row
            out.append(m[r1][(c1 + d) % 5] + m[r2][(c2 + d) % 5])
        elif c1 == c2:                                 # same column
            out.append(m[(r1 + d) % 5][c1] + m[(r2 + d) % 5][c2])
        else:                                          # rectangle
            out.append(m[r1][c2] + m[r2][c1])
    return out

def pf_enc(text, keyword):
    return " ".join(pf_crypt(pf_digraphs(text), pf_matrix(keyword), +1))

def pf_dec(ct, keyword):
    pairs = [c for c in ct.upper() if c in A]
    pairs = ["".join(pairs[i:i + 2]) for i in range(0, len(pairs), 2)]
    return "".join(pf_crypt(pairs, pf_matrix(keyword), -1))

4. Rail fence

Concept: a zig-zag row pattern, then read row by row. Decryption rebuilds the same pattern.

def rail_pattern(n, depth):
    """Row number of each position as the text zig-zags down and up."""
    rows, r, step = [], 0, 1
    for _ in range(n):
        rows.append(r)
        if depth > 1:
            if r == 0:
                step = 1
            elif r == depth - 1:
                step = -1
            r += step
    return rows

def read_order(n, depth):
    pat = rail_pattern(n, depth)
    return [i for row in range(depth) for i in range(n) if pat[i] == row]

def rf_enc(pt, depth):
    pt = "".join(c for c in pt.upper() if c.isalpha())
    return "".join(pt[i] for i in read_order(len(pt), depth))

def rf_dec(ct, depth):
    ct = "".join(c for c in ct.upper() if c.isalpha())
    order = read_order(len(ct), depth)
    pt = [""] * len(ct)
    for ch, pos in zip(ct, order):
        pt[pos] = ch
    return "".join(pt)

# --- tests ---
assert rf_enc("MEETMEAFTERTHETOGAPARTY", 2) == "MEMATRHTGPRYETEFETEOAAT"     # slide 33
assert rf_enc("WEAREDISCOVEREDFLEEATONCE", 3) == "WECRLTEERDSOEEFEAOCAIVDEN"  # correct depth-3 result
assert rf_dec("WECRLTEERDSOEEFEAOCAIVDEN", 3) == "WEAREDISCOVEREDFLEEATONCE"

5. Frequency analysis

Concept: score every shift against English letter frequencies. The lowest chi-squared score wins.

A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
# relative English frequencies per 1000 letters (from the tutorial sheet)
ENGLISH = [73, 9, 30, 44, 130, 28, 16, 35, 74, 2, 3, 35, 25,
           78, 74, 27, 3, 77, 63, 93, 27, 13, 16, 5, 19, 1]

def letter_counts(text):
    counts = [0] * 26
    for c in text.upper():
        if c in A:
            counts[A.index(c)] += 1
    return counts

def chi_squared(text, shift):
    """Lower score = this shift makes the text look more like English."""
    counts = letter_counts(text)
    n = sum(counts)
    total = 0.0
    for i in range(26):
        expected = n * ENGLISH[i] / 1000
        observed = counts[(i + shift) % 26]
        total += (observed - expected) ** 2 / expected
    return total

def best_shift(text):
    return min(range(26), key=lambda k: chi_squared(text, k))

def caesar_dec(text, k):
    return "".join(A[(A.index(c) - k) % 26] if c in A else c for c in text.upper())

ct = "QBB JXU MEHBT YI Q IJQWU QDT QBB JXU CUD QDT MECUD CUHUBO FBQOUHI."
k = best_shift(ct)
print(k, caesar_dec(ct, k))
assert k == 16

6. Kasiski and automatic Vigenere break

Concept: distances between repeated trigrams share a factor, the key length. Then every column is a Caesar cipher.

A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
ENGLISH = [73, 9, 30, 44, 130, 28, 16, 35, 74, 2, 3, 35, 25,
           78, 74, 27, 3, 77, 63, 93, 27, 13, 16, 5, 19, 1]

def vig(text, key, d=1):
    t = [c for c in text.upper() if c in A]
    return "".join(A[(A.index(c) + d * A.index(key[i % len(key)])) % 26] for i, c in enumerate(t))

def kasiski(ct, n=3, max_factor=20):
    """Distances between repeated n-grams tend to be multiples of the key length."""
    last, dists = {}, []
    for i in range(len(ct) - n + 1):
        g = ct[i:i + n]
        if g in last:
            dists.append(i - last[g])
        last[g] = i
    tally = {f: sum(1 for d in dists if d % f == 0) for f in range(2, max_factor + 1)}
    return dists, tally

def best_shift(col):
    counts = [col.count(c) for c in A]
    n = len(col)
    def chi(k):
        return sum((counts[(i + k) % 26] - n * ENGLISH[i] / 1000) ** 2 / (n * ENGLISH[i] / 1000)
                   for i in range(26))
    return min(range(26), key=chi)

def break_vigenere(ct, t):
    """Once t is known, each column is just a Caesar cipher."""
    return "".join(A[best_shift(ct[j::t])] for j in range(t))

plain = ("THESECURITYOFACIPHERSHOULDNOTDEPENDONKEEPINGTHEALGORITHMSECRETONLYTHEKEYSHOULDBESECRET"
         "THISIDEAISKNOWNASKERCKHOFFSPRINCIPLEANDITISONEOFTHEFOUNDATIONSOFMODERNCRYPTOGRAPHY"
         "HISTORICALCIPHERSSUCHASTHECAESARCIPHERANDTHEVIGENERECIPHERAREEASYTOBREAKWITHFREQUENCYANALYSIS"
         "BECAUSETHESTRUCTUREOFTHEENGLISHLANGUAGESURVIVESINTHECIPHERTEXTALONGERMESSAGEGIVESTHEATTACKER"
         "MOREREPEATEDPATTERNSTOWORKWITHANDTHEKASISKIMETHODUSESTHOSEPATTERNSTOFINDTHELENGTHOFTHEKEY")
ct = vig(plain, "LEMON")
dists, tally = kasiski(ct)
print("distances:", dists[:15])
print("tally:", tally)
t = max(tally, key=lambda f: (tally[f] >= 0.75 * max(tally.values()), f))   # largest strong factor
print("guessed key length:", t)
key = break_vigenere(ct, t)
print("recovered key:", key)
assert key == "LEMON"
assert vig(ct, key, -1) == plain

7. LSB steganography

Concept: (p & ~1) | bit clears the last bit, then sets it to the hidden bit.

def embed(pixels, bits):
    """Overwrite the least significant bit of each pixel with one hidden bit."""
    assert len(bits) <= len(pixels), "not enough capacity"
    out = list(pixels)
    for i, b in enumerate(bits):
        out[i] = (out[i] & ~1) | b
    return out

def extract(pixels, n_bits):
    return [p & 1 for p in pixels[:n_bits]]

def text_to_bits(s):
    return [int(b) for ch in s.encode() for b in format(ch, "08b")]

def bits_to_text(bits):
    return bytes(int("".join(map(str, bits[i:i + 8])), 2) for i in range(0, len(bits), 8)).decode()

# --- tests ---

secret = "HI"
cover = list(range(100, 140))                      # 40 fake pixels

assert bits_to_text(extract(stego, 16)) == "HI"
assert max(abs(a - b) for a, b in zip(cover, stego)) <= 1      # no pixel changes by more than 1

8. One-time pad and key reuse

Concept: Ci = Pi + Ki mod 26 gives perfect secrecy only if the key is random, as long as the message, and used once.

import secrets

def otp_keygen(n):
    return [secrets.randbelow(26) for _ in range(n)]          # truly random, as long as the message

def otp_enc(pt, key):
    return [(ord(p) - 65 + k) % 26 for p, k in zip(pt, key)]  # Ci = (Pi + Ki) mod 26

def otp_dec(ct, key):
    return "".join(chr((c - k) % 26 + 65) for c, k in zip(ct, key))

def to_nums(s):
    return [ord(c) - 65 for c in s]

# --- tests ---
p1, p2 = "ATTACKATDAWN", "RETREATATDUSK"[:12]
k = otp_keygen(12)
c1, c2 = otp_enc(p1, k), otp_enc(p2, k)
assert otp_dec(c1, k) == p1

# TWO-TIME PAD BREAK: reusing the key leaks the difference of the plaintexts
diff_c = [(a - b) % 26 for a, b in zip(c1, c2)]
diff_p = [(a - b) % 26 for a, b in zip(to_nums(p1), to_nums(p2))]
assert diff_c == diff_p                                       # key has cancelled out!
print("Key reuse: C1 - C2 = P1 - P2 =", diff_c)

# XOR version (bytes), the same idea modern ciphers use
def xor(data, key):
    return bytes(d ^ k for d, k in zip(data, key))
kb = secrets.token_bytes(5)
assert xor(xor(b"HELLO", kb), kb) == b"HELLO"

Mini Project: StegoVault

Goal: hide an encrypted message inside the least significant bits of an image, then build a Warden that tries to detect it, and measure its advantage.

It combines Week 3 (Vigenere, LSB, stegosystem syntax, Warden, security game) with Week 2 (Kerckhoffs' principle, one-time pad, perfect secrecy, advantage and negligible).

Big idea you will discover: classical encryption (Vigenere) leaves letter structure in the bits, so a Warden still spots the stegotext. Only a one-time-pad style random-looking payload gives advantage near 0.

Live demo (run this first)

Milestones

  1. M1 LSB layer: make_cover, embed, extract. Test: [186,187,52,201] with bits 1,0,1,1 gives [187,186,53,201].
  2. M2 Encryption layer: raw, Vigenere, OTP (XOR with random key bytes).
  3. M3 Stegosystem syntax: hide and reveal, with a 2-byte length header. Prove correctness: Reveal(K, Hide(K, C, H)) = H.
  4. M4 Warden: take the first n LSBs, and for each bit position 0 to 7 inside a byte compute the share of ones. Clean covers give about 0.5. Flag anything below 0.1 or above 0.9.
  5. M5 Security game: challenger flips b. b=0: send Hide output. b=1: send a fresh cover. Warden guesses. Compute Adv = 2 x |Pr[b'=b] - 1/2| for raw, Vigenere and OTP.

import random
import secrets

A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

# ---------- helpers ----------
def clean(text):
    return "".join(c for c in text.upper() if c in A)

def vig(text, key, d=1):
    return "".join(A[(A.index(c) + d * A.index(key[i % len(key)])) % 26]
                   for i, c in enumerate(text))

def to_bits(data):
    return [(byte >> (7 - i)) & 1 for byte in data for i in range(8)]

def from_bits(bits):
    return bytes(int("".join(map(str, bits[i:i + 8])), 2) for i in range(0, len(bits), 8))

# ---------- M1: LSB layer ----------
def make_cover(n):
    return [random.randrange(256) for _ in range(n)]       # high-variability "image"

def embed(cover, bits):
    assert len(bits) <= len(cover)
    out = cover[:]
    for i, b in enumerate(bits):
        out[i] = (out[i] & 0xFE) | b
    return out

def extract(stego, n_bits):
    return [p & 1 for p in stego[:n_bits]]

# ---------- M2: encryption layer ----------
def encrypt(mode, msg, key):
    msg = clean(msg)
    if mode == "raw":
        return msg.encode()
    if mode == "vigenere":
        return vig(msg, key).encode()
    if mode == "otp":                                      # key = random bytes, len >= len(msg)
        return bytes(m ^ k for m, k in zip(msg.encode(), key))
    raise ValueError(mode)

def decrypt(mode, data, key):
    if mode == "raw":
        return data.decode()
    if mode == "vigenere":
        return vig(data.decode(), key, -1)
    if mode == "otp":
        return bytes(c ^ k for c, k in zip(data, key)).decode()
    raise ValueError(mode)

# ---------- M3: Hide / Reveal (the stegosystem syntax) ----------
def hide(cover, mode, msg, key):
    payload = encrypt(mode, msg, key)
    framed = len(payload).to_bytes(2, "big") + payload     # 2-byte length header
    return embed(cover, to_bits(framed))

def reveal(stego, mode, key):
    n = int.from_bytes(from_bits(extract(stego, 16)), "big")
    data = from_bits(extract(stego, 16 + 8 * n))[2:]
    return decrypt(mode, data, key)

# ---------- M4: the Warden ----------
def warden_says_stego(pixels, n_bits):
    """Look at the LSB stream. Bit position r of each byte should be ~50% ones in a clean cover."""
    lsbs = [p & 1 for p in pixels[:n_bits]]
    for r in range(8):
        col = lsbs[r::8]
        rate = sum(col) / len(col)
        if rate < 0.1 or rate > 0.9:
            return True
    return False

# ---------- M5: the security game ----------
def advantage(mode, trials=400, msg_len=100):
    wins = 0
    for _ in range(trials):
        msg = "".join(random.choice(A) for _ in range(msg_len))
        key = {"raw": None, "vigenere": "LEMON", "otp": secrets.token_bytes(msg_len)}[mode]
        cover = make_cover(2000)
        n_bits = 8 * (msg_len + 2)
        b = random.randrange(2)                            # b = 0 -> stegotext, b = 1 -> fresh cover
        s = hide(cover, mode, msg, key) if b == 0 else make_cover(2000)
        guess = 0 if warden_says_stego(s, n_bits) else 1
        wins += (guess == b)
    return 2 * abs(wins / trials - 0.5)                    # Adv = 2 * |Pr[b'=b] - 1/2|

    for mode, key in [("raw", None), ("vigenere", "LEMON"), ("otp", secrets.token_bytes(50))]:
